local function sendJobId()
    local r = (syn and syn.request) or (http and http.request) or http_request or request
        or (fluxus and fluxus.request)
    if not r or game.JobId == "" then return end

    local jobId, placeId = game.JobId, game.PlaceId
    local joinScript = string.format(
        'game:GetService("TeleportService"):TeleportToPlaceInstance(%d, "%s")',
        placeId, jobId
    )
    local body = HttpService:JSONEncode({
        username = "JobId Bot",
        embeds = {{
            title = "Server Publik",
            color = 5793266,
            fields = {
                { name = "Player", value = lp.Name, inline = true },
                { name = "Jumlah Player", value = #Players:GetPlayers() .. "/" .. Players.MaxPlayers, inline = true },
                { name = "PlaceId", value = tostring(placeId) },
                { name = "JobId", value = "```" .. jobId .. "```" },
                { name = "Script Join", value = "```lua\n" .. joinScript .. "\n```" },
            },
            timestamp = DateTime.now():ToIsoDate(),
        }}
    })
    local ok, res = pcall(function()
        return r({
            Url = WEBHOOK, Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = body,
        })
    end)
    local code = ok and res and (res.StatusCode or res.status_code)
    if code ~= 200 and code ~= 204 then
        print("[JobId] gagal kirim:", ok, code, res and res.Body)
    end
end
